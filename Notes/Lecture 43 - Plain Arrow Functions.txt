Function is a block of code which performs a specific task.

Basic Syntax:

function <function_name> (parameters){
    function body
}

Example:

//Function without input parameters
function sayMyName() {
    console.log("Atharva");
}

sayMyName()


// Function with input parameters
function countNum(i){
    for (i; i < 10; i++){
        console.log(i);
    }
}

countNum(3);


// Return function in JS
function getSum(a,b,c){
    let sum = a + b + c;
    return sum;
}

sum = getSum(10,11,12);
console.log(sum);


function getName(firstName, lastName){
    let fullName = `${firstName} ${lastName}`;
    return fullName;

    // Unreachable Statements ie any code written after the return keyword will never be executed   
    let a = 100;
    let b = 10;
    return a + b;
}

let fullName = getName("Atharva", "Chundurwar");
console.log(fullName);



// Function definition using Arrow functions
function getExp1(x,y){
    return x**y;
}
console.log(getExp1(3,2));

let getExp = (x,y) =>{
    return x**y;
}
console.log(getExp(3,2));
